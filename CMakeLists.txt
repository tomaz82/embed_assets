if(NOT DEFINED EMBED_INPUT_DIR)
    set(EMBED_INPUT_DIR "${CMAKE_CURRENT_LIST_DIR}/input")
endif()
if(NOT DEFINED EMBED_OUTPUT_DIR)
    set(EMBED_OUTPUT_DIR "${CMAKE_CURRENT_LIST_DIR}/output")
endif()
if(NOT DEFINED EMBED_NAMESPACE)
    set(EMBED_NAMESPACE "Embedded")
endif()

file(MAKE_DIRECTORY "${EMBED_OUTPUT_DIR}")

function(sanitize_symbol INPUT OUTVAR)
    string(TOLOWER "${INPUT}" _clean)
    string(REGEX REPLACE "[^a-z0-9_]" "_" _clean "${_clean}")
    set(${OUTVAR} "${_clean}" PARENT_SCOPE)
endfunction()

function(path_to_namespace RELPATH OUTVAR)
    if("${RELPATH}" STREQUAL "" OR "${RELPATH}" STREQUAL ".")
        set(${OUTVAR} "" PARENT_SCOPE)
    else()
        string(REPLACE "\\" "/" _norm "${RELPATH}")
        string(REPLACE "/" ";" _parts "${_norm}")
        set(_ns "")
        foreach(_p IN LISTS _parts)
            sanitize_symbol("${_p}" _clean)
            if(_ns STREQUAL "")
                set(_ns "${_clean}")
            else()
                set(_ns "${_ns}::${_clean}")
            endif()
        endforeach()
        set(${OUTVAR} "${_ns}" PARENT_SCOPE)
    endif()
endfunction()

function(generate_header OUT_H OUT_NAMESPACE OUT_BASE)
    file(WRITE "${OUT_H}" "#pragma once\n\n")
    if(NOT "${OUT_NAMESPACE}" STREQUAL "")
        file(APPEND "${OUT_H}" "namespace ${EMBED_NAMESPACE} {\nnamespace ${OUT_NAMESPACE} {\nnamespace ${OUT_BASE} {\n")
    else()
        file(APPEND "${OUT_H}" "namespace ${EMBED_NAMESPACE} {\nnamespace ${OUT_BASE} {\n")
    endif()
    file(APPEND "${OUT_H}" "extern const unsigned char* data();\n")
    file(APPEND "${OUT_H}" "extern unsigned int size();\n")
    if(NOT "${OUT_NAMESPACE}" STREQUAL "")
        file(APPEND "${OUT_H}" "} // ${OUT_BASE}::\n} // ${OUT_NAMESPACE}::\n} // ${EMBED_NAMESPACE}::\n")
    else()
        file(APPEND "${OUT_H}" "} // ${OUT_BASE}::\n} // ${EMBED_NAMESPACE}::\n")
    endif()
endfunction()

function(generate_source IN_FILE OUT_CPP OUT_H OUT_NAMESPACE OUT_BASE)
    file(READ "${IN_FILE}" _file_hex HEX)
    file(SIZE "${IN_FILE}" _file_size)
    string(LENGTH "${_file_hex}" _hex_len)
    file(RELATIVE_PATH _rel_header "${EMBED_OUTPUT_DIR}" "${OUT_H}")
    file(WRITE "${OUT_CPP}" "\n")
    if(NOT "${OUT_NAMESPACE}" STREQUAL "")
        file(APPEND "${OUT_CPP}" "namespace ${EMBED_NAMESPACE} {\nnamespace ${OUT_NAMESPACE} {\nnamespace ${OUT_BASE} {\n")
    else()
        file(APPEND "${OUT_CPP}" "namespace ${EMBED_NAMESPACE} {\nnamespace ${OUT_BASE} {\n")
    endif()
    file(APPEND "${OUT_CPP}" "static constexpr unsigned char _data[] = {\n")
    set(_line "")
    math(EXPR _pairs "${_hex_len} / 2 - 1")
    if(_pairs GREATER_EQUAL 0)
        foreach(_i RANGE 0 ${_pairs})
            math(EXPR _pos "${_i} * 2")
            string(SUBSTRING "${_file_hex}" ${_pos} 2 _byte_hex)
            string(TOUPPER "${_byte_hex}" _byte_hex)
            set(_line "${_line}0x${_byte_hex}, ")
            math(EXPR _mod "${_i} % 16")
            if(_mod EQUAL 11)
                file(APPEND "${OUT_CPP}" "    ${_line}\n")
                set(_line "")
            endif()
        endforeach()
    endif()
    if(NOT _line STREQUAL "")
        file(APPEND "${OUT_CPP}" "    ${_line}\n")
    endif()
    file(APPEND "${OUT_CPP}" "};\nstatic constexpr unsigned int _size = ${_file_size};\n\n")
    file(APPEND "${OUT_CPP}" "const unsigned char* data() { return _data; }\n")
    file(APPEND "${OUT_CPP}" "unsigned int size() { return _size; }\n")
    if(NOT "${OUT_NAMESPACE}" STREQUAL "")
        file(APPEND "${OUT_CPP}" "} // ${OUT_BASE}::\n} // ${OUT_NAMESPACE}::\n} // ${EMBED_NAMESPACE}::\n")
    else()
        file(APPEND "${OUT_CPP}" "} // ${OUT_BASE}::\n} // ${EMBED_NAMESPACE}::\n")
    endif()
endfunction()

set(_MASTER_HEADER "${EMBED_OUTPUT_DIR}/embedded_files.h")
file(WRITE "${_MASTER_HEADER}" "#pragma once\n\n")

function(embed_file ABS_INPUT_PATH REL_PATH)
    get_filename_component(_fname "${REL_PATH}" NAME)
    get_filename_component(_base "${_fname}" NAME_WE)
    get_filename_component(_ext "${_fname}" EXT)
    string(SUBSTRING "${_ext}" 1 -1 _ext_no_dot)
    sanitize_symbol("${_base}" _base_san)
    sanitize_symbol("${_ext_no_dot}" _ext_san)
    set(_base_with_ext "${_base_san}_${_ext_san}")
    get_filename_component(_rel_dir "${REL_PATH}" PATH)
    if(_rel_dir STREQUAL ".")
        set(_rel_dir "")
    endif()
    path_to_namespace("${_rel_dir}" _ns_frag)
    if(NOT "${_rel_dir}" STREQUAL "")
        set(_OUT_SUBDIR "${EMBED_OUTPUT_DIR}/${_rel_dir}")
    else()
        set(_OUT_SUBDIR "${EMBED_OUTPUT_DIR}")
    endif()
    file(MAKE_DIRECTORY "${_OUT_SUBDIR}")
    set(_out_h "${_OUT_SUBDIR}/${_base_with_ext}.h")
    set(_out_cpp "${_OUT_SUBDIR}/${_base_with_ext}.cpp")
    file(RELATIVE_PATH _rel_header "${EMBED_OUTPUT_DIR}" "${_out_h}")
    file(APPEND "${_MASTER_HEADER}" "#include \"${_rel_header}\"\n")

    set(_generate TRUE)
    if(EXISTS "${_out_h}" AND EXISTS "${_out_cpp}")
        file(TIMESTAMP "${ABS_INPUT_PATH}" _input_time UTC)
        file(TIMESTAMP "${_out_h}" _h_time UTC)
        file(TIMESTAMP "${_out_cpp}" _cpp_time UTC)
        if(_input_time STRLESS _h_time AND _input_time STRLESS _cpp_time)
            message(STATUS "Skipping (up-to-date): ${REL_PATH}")
            set(_generate FALSE)
        endif()
    endif()

    if(_generate)
        message(STATUS "Generating: ${REL_PATH}")
        generate_header("${_out_h}" "${_ns_frag}" "${_base_with_ext}")
        generate_source("${ABS_INPUT_PATH}" "${_out_cpp}" "${_out_h}" "${_ns_frag}" "${_base_with_ext}")
    endif()
endfunction()

function(embed_folder CURRENT_SRC CURRENT_DST REL_PREFIX)
    file(GLOB _children "${CURRENT_SRC}/*")
    foreach(_ch IN LISTS _children)
        if(IS_DIRECTORY "${_ch}")
            get_filename_component(_dname "${_ch}" NAME)
            if("${REL_PREFIX}" STREQUAL "")
                set(_new_rel "${_dname}")
            else()
                set(_new_rel "${REL_PREFIX}/${_dname}")
            endif()
            embed_folder("${_ch}" "${CURRENT_DST}/${_dname}" "${_new_rel}")
        else()
            if("${REL_PREFIX}" STREQUAL "")
                get_filename_component(_name "${_ch}" NAME)
                set(_rel_path "${_name}")
            else()
                get_filename_component(_name "${_ch}" NAME)
                set(_rel_path "${REL_PREFIX}/${_name}")
            endif()
            embed_file("${_ch}" "${_rel_path}")
        endif()
    endforeach()
endfunction()

message(STATUS "")
message(STATUS "Embedding folder: ${EMBED_INPUT_DIR}")
embed_folder("${EMBED_INPUT_DIR}" "${EMBED_OUTPUT_DIR}" "")
message(STATUS "")
